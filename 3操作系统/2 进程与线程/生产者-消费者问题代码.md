```c
semaphore empty = n; // 同步信号量
semaphore full = 0; // 同步信号量
semaphore mutex = 1; // 互斥信号量

// 生产者
producer() {
	while (1) {
		p(empty); // 申请一个生产者可用数量
		p(mutex); // 互斥使用缓冲区
		把产品放入缓冲区
		v(mutex); // 释放缓冲区占用
		v(full); // 增加一个消费者可用数量
	}
}

// 消费者
consumer() {
	while (1) {
		p(full); // 申请一个消费者可用数量，若该信号量≤0，则阻塞，不进行消费
		p(mutex); // 互斥使用缓冲区
		从缓冲区中取出一个产品
		v(mutex); // 释放缓冲区占用
		v(empty); // 增加一个缓冲区/生产者可用数量
	}
}
```
