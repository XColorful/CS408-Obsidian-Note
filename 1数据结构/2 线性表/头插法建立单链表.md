```c
LinkList List_HeadInsert(LinkList &L) { //逆向建立单链表
	LNode *s;
	int x;
	L = (LinkList)malloc(sizeof(LNode)); //创建头结点
	L->next = NULL; //初始为空链表
	scanf("%d", &x); //输入节点的值
	while (x != 9999) { //输入9999表示结束
		s = (LNode *)malloc(sizeof(LNode)); //创建新节点
		s->data = x;
		s->next = L->next;
		L->next = s; //将新节点插入表中，L为头指针
		scanf("%d", &x);
	}
	return L;
}
```
> 插入的时间为$O(1)$；时间复杂度为$O(n)$
