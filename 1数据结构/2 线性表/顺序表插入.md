```c
bool ListInsert(SqList &L, int i, ElemType e) {
	if (i < 1 || i > L.length + 1) //判断i的范围是否有效 (1~n+1)
		return false;
	if (L.length >= Maxsize) //当前存储空间已满，不能插入
		return false;
	for (int j = L.length; j >= i; j--) //将第i个元素及以后的元素后移
		L.data[j] = L.data[j - 1];
	L.data[i - 1] = e; //在位置i处放入e
	L.length++;
	return true;
}
```
> 📢在第i个位置插入一个元素，需要移动$n-i+1$【$n+1-(i+1)+1$】个元素（此时$1≤i≤n+1$，i从1开始）
