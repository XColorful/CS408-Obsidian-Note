```c
LinkList List_TailInsert(LinkList &L) { //正向建立单链表
	int x;
	L = (LinkList)malloc(sizeof(LNode));
	LNode *s, *r = L; // r为表尾指针
	scanf("%d", &x); //输入节点的值
	while (x != 9999) { //输入9999表示结束
		s = (LNode *)malloc(sizeof(LNode)); //创建新节点
		s->data = x;
		r->next = s;
		r = s; // r指向新的表尾节点
		scanf("%d", &x);
	}
	r->next = NULL; //尾节点指针置空
	return L;
}
```
> 时间复杂度为$O(n)$
