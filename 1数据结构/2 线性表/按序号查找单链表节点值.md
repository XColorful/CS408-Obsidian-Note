```c
LNode *GetElem(LinkList L, int i) {
	int j = 1; //计数，初始为1
	LNode *p = L->next; //第1个节点指针赋给p
	if (i == 0)
		return L; //若i=0，则返回头结点
	if (i < 1)
		return NULL; //若i无效，则返回NULL
	while (p && j < i) { //从第i个节点开始找，查找第i个节点
		p = p->next;
		j++;
	}
	return p; //返回第i个节点的指针，若i大于表长，则返回NULL
}
```
> 时间复杂度为$O(n)$
